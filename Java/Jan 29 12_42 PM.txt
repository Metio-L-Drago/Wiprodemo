import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.Map;

public class Main {

	public static void main(String[] args) {
		
		Courses c1=new Courses("java" ,4999);
		Courses c2=new Courses("js" ,3999);
		Courses c3=new Courses("python" ,5999);
		Courses c4=new Courses("c++" ,1999);
		
	
		// KEYs are always unique
		HashMap<Integer,Courses > map=new HashMap<>();
		map.put(1,c1);
		map.put(2,c2);
		map.put(3, c3);
		map.put(4, c3);
	  
		   for( Map.Entry<Integer, Courses>   obj  :  map.entrySet())
		   {
			   
			   
			   Courses c=obj.getValue();
			    System.out.println(obj.getKey()+" - "+c.lang+" - "+c.price);
			  
		   }
		   
	
	}

}

{
	String lang;
	int price;
	
	public Courses(String lang, int price) {
		this.lang = lang;
		this.price = price;
	}
